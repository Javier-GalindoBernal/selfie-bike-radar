import { NextRequest, NextResponse } from "next/server";
import { exchangeCode, syncMember, baseUrl } from "@/lib/strava";
import { store } from "@/lib/store";

export async function GET(req: NextRequest) {
  const code = req.nextUrl.searchParams.get("code");
  const memberId = req.nextUrl.searchParams.get("state");
  const back = (q: string) => NextResponse.redirect(`${baseUrl()}/share?${q}`);
  if (!code || !memberId) return back("strava=denied");
  try {
    await exchangeCode(memberId, code);
    const m = await store.getMember(memberId);
    if (m) await syncMember(m);
    return back("strava=ok");
  } catch (e) {
    console.error(e);
    return back("strava=error");
  }
}
