import { NextRequest, NextResponse } from "next/server";
import { authorizeUrl } from "@/lib/strava";
import { store } from "@/lib/store";

export async function GET(req: NextRequest) {
  const id = req.nextUrl.searchParams.get("member") ?? "";
  if (!process.env.STRAVA_CLIENT_ID) {
    return NextResponse.json({ error: "Falta STRAVA_CLIENT_ID en .env.local" }, { status: 500 });
  }
  if (!(await store.getMember(id))) {
    return NextResponse.json({ error: "Miembro no existe" }, { status: 404 });
  }
  return NextResponse.redirect(authorizeUrl(id));
}
